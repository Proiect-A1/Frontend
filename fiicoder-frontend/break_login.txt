// 1. Creăm payload-ul cu datele de care are nevoie AuthContext-ul tău
const mockPayload = {
  sub: "admin_local",
  userId: "123-admin",
  role: "ADMIN",
  iat: Math.floor(Date.now() / 1000),
  exp: Math.floor(Date.now() / 1000) + (60 * 60 * 24) // Expiră în 24 de ore
};

// 2. Transformăm payload-ul în Base64 (exact cum se așteaptă atob să îl găsească)
const base64Payload = btoa(JSON.stringify(mockPayload));

// 3. Asamblăm un JWT fals (header.payload.semnătura). Frontend-ul ignoră header-ul și semnătura.
const fakeToken = `dummy_header.${base64Payload}.dummy_signature`;

// 4. Îl salvăm în cheia pe care o caută aplicația ta și dăm refresh
localStorage.setItem('fiicoder_jwt', fakeToken);
window.location.reload();

console.log("Acum ești admin! 🚀");
